Total Transactions = COUNTROWS(fact_transactions)
Total Transaction Amount = SUM(fact_transactions[amount_inr])
Average Transaction Value = DIVIDE([Total Transaction Amount],[Total Transactions])
Successful Transactions = CALCULATE([Total Transactions],fact_transactions[status]="SUCCESS")
Failed Transactions = CALCULATE([Total Transactions],fact_transactions[status]="FAILED")
Failed Transaction Rate = DIVIDE([Failed Transactions],[Total Transactions])
Pending Transactions = CALCULATE([Total Transactions],fact_transactions[status]="PENDING")
Pending Transaction Rate = DIVIDE([Pending Transactions],[Total Transactions])
Chargeback Count = DISTINCTCOUNT(fact_chargebacks[complaint_id])
Chargeback Amount = SUM(fact_chargebacks[disputed_amount_inr])
Chargeback-to-Transaction Ratio = DIVIDE([Chargeback Count],[Total Transactions])
KYC Completion Rate = DIVIDE(CALCULATE(DISTINCTCOUNT(dim_customers[user_id]),dim_customers[kyc_status]="VERIFIED"),DISTINCTCOUNT(dim_customers[user_id]))
High Risk Users = CALCULATE(DISTINCTCOUNT(dim_customers[user_id]),dim_customers[risk_segment]="HIGH")
